```sql
\echo '# post10 이름 열 채움 · 이름→코드 · 유사명 · SELECT only'
SELECT now() AS run_at;
\echo '## 이름 열 채움'
SELECT 'stock_info', count(*), count(stock_name) FILTER (WHERE stock_name<>'') FROM stock_info;
SELECT 'stock_industry', count(*), count(stock_name) FILTER (WHERE stock_name<>'') FROM stock_industry;
SELECT 'candidate_stocks', count(*), count(stock_name) FILTER (WHERE stock_name<>'') FROM candidate_stocks;
SELECT 'stock_sector', count(*), count(stock_name) FILTER (WHERE stock_name<>'') FROM stock_sector;
SELECT 'dart_corp_code', count(*), count(corp_name) FILTER (WHERE corp_name<>''), max(updated_at) FROM dart_corp_code;
\echo '## 이름 LIKE 매칭 (4표)'
WITH q(nm) AS (VALUES ('한컴위드'),('라온시큐어'),('범한퓨얼셀'),('서산'),('코데즈컴바인'),('샌즈랩'),('빛샘전자'),('성호전자'),('HT로보틱스'),('한켐'),('뷰노'),('우리로'))
SELECT q.nm, t.src, t.stock_code, t.stock_name FROM q
JOIN (
  SELECT 'stock_info' src, stock_code, stock_name FROM stock_info
  UNION ALL SELECT 'stock_industry', stock_code, stock_name FROM stock_industry
  UNION ALL SELECT DISTINCT 'candidate_stocks', stock_code, stock_name FROM candidate_stocks
  UNION ALL SELECT 'stock_sector', stock_code, stock_name FROM stock_sector
) t ON t.stock_name LIKE '%'||q.nm||'%'
ORDER BY q.nm, t.stock_code, t.src;
\echo '## 부분명·별칭 LIKE — 유사명 점검'
WITH q(nm) AS (VALUES ('한컴'),('라온'),('범한'),('코데즈'),('샌즈'),('빛샘'),('성호'),('로보틱스'),('에이치티'),('HT'),('한켐'),('뷰노'),('우리'))
SELECT q.nm, t.stock_code, t.stock_name FROM q
JOIN (SELECT DISTINCT stock_code, stock_name FROM stock_info UNION SELECT DISTINCT stock_code, stock_name FROM stock_industry UNION SELECT DISTINCT stock_code, stock_name FROM stock_sector) t
ON t.stock_name LIKE '%'||q.nm||'%' ORDER BY 1,2;
```

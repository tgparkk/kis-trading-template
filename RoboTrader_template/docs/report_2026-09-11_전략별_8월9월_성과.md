## SQL
```sql
-- 실현(매도 기준 · net 은 공식 적용)
WITH s AS (SELECT to_char(timestamp,'YYYY-MM') ym, strategy, quantity*price sell_amt, profit_loss g,
  profit_loss - 0.00015*(quantity*price - profit_loss) - 0.00195*(quantity*price) n
  FROM virtual_trading_records WHERE action='SELL' AND timestamp>='2026-08-01')
SELECT ym, strategy, count(*), sum(CASE WHEN g>0 THEN 1 ELSE 0 END), round(sum(g)), round(sum(n)), round(sum(sell_amt)),
       round(100*sum(g)/nullif(sum(sell_amt-g),0),2) FROM s GROUP BY 1,2 ORDER BY 1,2;
-- 자본
SELECT strategy, trade_date, equity FROM paper_strategy_equity WHERE trade_date IN ('2026-07-31','2026-08-31','2026-09-11') ORDER BY 1,2;
-- 지수
SELECT index_code, date, close FROM index_daily WHERE date IN ('2026-07-31','2026-08-31','2026-09-11') ORDER BY 1,2;
```

-- 8월 창: 매도 timestamp >= '2026-08-01' AND < '2026-09-01' (세션 tz KST) · 자본은 trade_date 07-31 / 08-31

SELECT a.strategy, round(100*(a.equity/b.equity-1),2)
FROM paper_strategy_equity a JOIN paper_strategy_equity b ON b.strategy=a.strategy AND b.trade_date='2026-07-31'
WHERE a.trade_date='2026-08-31';
-- 전략별 8월 매도 집계
SELECT strategy, count(*), round(100.0*count(*) FILTER (WHERE profit_loss>0)/count(*),1), sum(profit_loss),
  round(100.0*count(*) FILTER (WHERE reason LIKE '%손절%')/count(*),1)
FROM virtual_trading_records WHERE is_test=true AND action='SELL' AND timestamp>='2026-08-01' AND timestamp<'2026-09-01' GROUP BY 1;
-- 중앙값 매매 = 전략별 profit_loss 오름차순 (n+1)/2 번째 (동률은 id)
